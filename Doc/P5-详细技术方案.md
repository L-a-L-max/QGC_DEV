### 缩放算法

```cpp
float applyJoystickCurve(float input, float deadzone, float expo) {
    float sign = (input >= 0) ? 1.0f : -1.0f;
    float abs_in = fabsf(input);
    
    // 死区
    if (abs_in < deadzone) return 0.0f;
    
    // 重映射到 [0,1]（死区后的有效范围）
    float normalized = (abs_in - deadzone) / (1.0f - deadzone);
    
    // 指数曲线: output = (1-expo)*normalized + expo*normalized^3
    float curved = (1.0f - expo) * normalized + expo * normalized * normalized * normalized;
    
    return sign * curved;
}

// 速度缩放: maxSpeed / MPC_XY_VEL_MAX_DEFAULT (12.0)
float speedScale = maxSpeed / 12.0f;
roll  = applyJoystickCurve(roll,  deadzone, expo) * speedScale;
pitch = applyJoystickCurve(pitch, deadzone, expo) * speedScale;
```

### NED 坐标转换

```cpp
// 经纬度 → NED（以 Home 位置为原点）
void geoToNED(double lat, double lon, double alt,
              double homeLat, double homeLon, double homeAlt,
              float &north, float &east, float &down) {
    constexpr double R = 6371000.0; // 地球半径
    double dLat = qDegreesToRadians(lat - homeLat);
    double dLon = qDegreesToRadians(lon - homeLon);
    double cosHomeLat = cos(qDegreesToRadians(homeLat));
    
    north = static_cast<float>(dLat * R);
    east  = static_cast<float>(dLon * R * cosHomeLat);
    down  = static_cast<float>(homeAlt - alt); // NED: down is negative altitude
}
```

### 航点数据结构

```cpp
struct DDSWaypoint {
    double latitude;      // WGS84 纬度
    double longitude;     // WGS84 经度
    float  altitudeRel;   // 相对 Home 高度 (m)
    float  speed;         // 最大水平速度 (m/s), ≤0 = 使用默认
    float  heading;       // 目标航向 (°), NaN = 不控制
    float  hoverTime;     // 到达后悬停时间 (s), 0 = 不悬停
```

### 状态机

```
          ┌──────────┐
          │  IDLE     │ ← 初始状态 / 任务完成后
          └────┬─────┘
    startMission()
               │
          ┌────▼─────┐
          │ RUNNING   │──── pauseMission() ───→ PAUSED
          │ (飞向WP_n)│                           │
          └────┬─────┘ ◄── resumeMission() ──────┘
               │
       到达当前航点
               │
          ┌────▼─────┐
          │ HOVERING  │ 悬停 hoverTime 秒
          └────┬─────┘
               │ 悬停完成 / hoverTime=0
               │
       ┌───────▼───────┐
       │ 还有下一航点?  │
       ├── Yes ──→ RUNNING (n+1)
       └── No  ──→ END_ACTION
                      │
              ┌───────▼───────┐
              │ endAction:     │
              │ 0=Hover 1=RTL │
              │ 2=Land        │
              └───────────────┘
```

### 关键接口

```cpp
class DDSMissionManager : public QObject {
    Q_OBJECT
    Q_PROPERTY(State state READ state NOTIFY stateChanged)
    Q_PROPERTY(int currentWaypointIndex READ currentWaypointIndex NOTIFY currentWaypointChanged)
    Q_PROPERTY(int waypointCount READ waypointCount NOTIFY waypointsChanged)

    Q_ENUM(EndAction)

    Q_INVOKABLE void addWaypoint(double lat, double lon, float alt, float speed, float heading, float hover);
    Q_INVOKABLE void removeWaypoint(int index);
    Q_INVOKABLE void clearWaypoints();
    Q_INVOKABLE void startMission();
    Q_INVOKABLE void pauseMission();
    Q_INVOKABLE void resumeMission();
    Q_INVOKABLE void stopMission();

    // Save/Load
    Q_INVOKABLE bool saveMission(const QString &filePath);
    Q_INVOKABLE bool loadMission(const QString &filePath);

    void stateChanged();
    void currentWaypointChanged();
```

### 到达判定

订阅 `fmu/out/vehicle_global_position`，持续检查：

```cpp
void DDSMissionManager::_checkArrival() {
    double dist = _currentPos.distanceTo(_waypoints[_currentIndex].coordinate());
    double altDiff = fabs(_currentAlt - _targetAltAMSL);
    
    if (dist < _arrivalRadiusH && altDiff < _arrivalRadiusV) {
        // 到达！
        emit waypointReached(_currentIndex);
        if (_waypoints[_currentIndex].hoverTime > 0) {
            _state = Hovering;
            _hoverTimer.start(_waypoints[_currentIndex].hoverTime * 1000);
        } else {
            _advanceToNext();
        }
    }
}
```

默认到达半径：水平 2.0m，垂直 1.0m。

## 六、P5-5: 任务结束动作

在 DDSMissionManager 中内置：

| 动作 | 实现 |
|------|------|
| **悬停 (Hover)** | 最后一个航点到达后停止发送 goto，无人机保持位置 |
| **返航 (RTL)** | 发送 `MAV_CMD_NAV_RETURN_TO_LAUNCH` (cmd 20) |
| **降落 (Land)** | 发送 `MAV_CMD_NAV_LAND` (cmd 21) |

UI 中用下拉框选择，默认"悬停"。

## 七、P5-6: 轨迹显示修复

### 问题

DDS 车辆 `_updateArmed` 中 `armed` 状态变化时不调用 `_trajectoryPoints->start()`，因为 DDS 路径直接设置 `_vehicle->_armed` 而不走 `_updateArmed()`。

### 修复

在 `DDSDataInjector::_updateVehicleState()` 的 armed 变化逻辑中，
添加 `_vehicle->_trajectoryPoints` 的 start/stop 调用：

```cpp
if (_vehicle->_armed != armed) {
    _vehicle->_armed = armed;
    emit _vehicle->armedChanged(armed);
    // 同步 trajectory 跟踪
    if (armed) {
        _vehicle->trajectoryPoints()->start();
    } else {
        _vehicle->trajectoryPoints()->stop();
    }
}
```
